local function GetCharacter()
    return LocalPlayer.Character
end

local function GetHumanoid()
    local char = GetCharacter()
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function GetRootPart()
    local char = GetCharacter()
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function DisableCollisions()
    local char = GetCharacter()
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide then
            part.CanCollide = false
        end
    end
end

-- Role Detection
GetMurderer = function()
    for _, player in ipairs(Players:GetPlayers()) do
        if player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife") then
            return player
        end
        if player.Character and player.Character:FindFirstChild("Knife") then
            return player
        end
    end
end

GetSheriff = function()
    for _, player in ipairs(Players:GetPlayers()) do
        if player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Gun") then
            return player
        end
        if player.Character and player.Character:FindFirstChild("Gun") then
            return player
        end
    end
end

-- Teleports
local function GetMapContainer()
    for _, child in ipairs(Workspace:GetChildren()) do
        if child:FindFirstChild("CoinContainer") then
            return child
        end
    end
end

TeleportToLobby = function()
    if Workspace:FindFirstChild("Lobby") and LocalPlayer.Character then
        LocalPlayer.Character:MoveTo(Workspace.Lobby.WorldPivot.Position)
    end
end

TeleportToMap = function()
    local map = GetMapContainer()
    if not map then return end
    local spawns = map:FindFirstChild("Spawns")
    if spawns then
        local children = spawns:GetChildren()
        if #children > 0 and LocalPlayer.Character then
            LocalPlayer.Character:MoveTo(children[math.random(1, #children)].Position)
        end
    end
end

local function TeleportToSafespot()
    local root = GetRootPart()
    if not root then return end
    if Workspace:FindFirstChild("safespot") then
        root.CFrame = CFrame.new(-106, 87.5, 2)
    else
        local part = Instance.new("Part")
        part.Name = "safespot"
        part.Size = Vector3.new(25, 3, 25)
        part.Position = Vector3.new(-106, 83, 2)
        part.Anchored = true
        part.CanCollide = true
        part.Transparency = 0
        part.Parent = Workspace
        task.wait()
        for i = 1, 3 do
            local r = GetRootPart()
            if r then
                r.CFrame = CFrame.new(-106, 87.5, 2)
                task.wait(0.1)
            end
        end
    end
end

-- Kill Everyone As Murderer
KillEveryoneAsMurderer = function()
    local char = GetCharacter()
    if not char then return end
    local knife = char:FindFirstChild("Knife")
    if not knife then
        local bpKnife = LocalPlayer.Backpack:FindFirstChild("Knife")
        if not bpKnife then return end
        local hum = GetHumanoid()
        if hum then hum:EquipTool(bpKnife) end
    end

    local myRoot = GetRootPart()
    if not myRoot then return end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local tRoot = player.Character:FindFirstChild("HumanoidRootPart")
            if tRoot then
                tRoot.Anchored = true
                tRoot.CFrame = myRoot.CFrame + (myRoot.CFrame.LookVector * 1)
            end
        end
    end

    pcall(function()
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Knife") then
            LocalPlayer.Character.Knife.Stab:FireServer("Slash")
        end
    end)
end

-- Fling System
FlingPlayers = function(targetNames)
    local allMode = false
    local function findTarget(input)
        local lower = input:lower()
        if lower == "all" or lower == "others" then
            allMode = true
            return nil
        end
        if lower == "random" then
            local list = {}
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer then table.insert(list, p) end
            end
            if #list == 0 then return nil end
            return list[math.random(#list)]
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                if p.Name:lower():match("^" .. lower) or p.DisplayName:lower():match("^" .. lower) then
                    return p
                end
            end
        end
    end

    local function flingTarget(targetPlayer)
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = hum and hum.RootPart
        if not char or not hum or not root then return end

        local targetChar = targetPlayer.Character
        if not targetChar then return end
        local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
        local targetRoot = targetHum and targetHum.RootPart
        local targetHead = targetChar:FindFirstChild("Head")

        local genv = (typeof(getgenv) == "function" and getgenv()) or _G
        if root.Velocity.Magnitude < 50 then
            genv.OldPos = root.CFrame
        end

        if targetHum and targetHum.Sit and not allMode then return end
        if targetHead then
            Workspace.CurrentCamera.CameraSubject = targetHead
        elseif targetHum and targetRoot then
            Workspace.CurrentCamera.CameraSubject = targetHum
        end

        if not targetChar:FindFirstChildWhichIsA("BasePart") then return end

        local function setFlingCFrame(targetPart, offsetCFrame, angleCFrame)
            root.CFrame = CFrame.new(targetPart.Position) * offsetCFrame * angleCFrame
            pcall(function()
                char:SetPrimaryPartCFrame(CFrame.new(targetPart.Position) * offsetCFrame * angleCFrame)
            end)
            root.Velocity = Vector3.new(90000000, 900000000, 90000000)
            root.RotVelocity = Vector3.new(900000000, 900000000, 900000000)
        end

        local function doFling(targetPart)
            local startTime = tick()
            local rot = 0
            while root and targetHum do
                if targetPart.Velocity.Magnitude < 50 then
                    rot = rot + 100
                    setFlingCFrame(targetPart, CFrame.new(0, 1.5, 0) + targetHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(rot), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, 0) + targetHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(rot), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(2.25, 1.5, -2.25) + targetHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(rot), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(-2.25, -1.5, 2.25) + targetHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(rot), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, 1.5, 0) + targetHum.MoveDirection, CFrame.Angles(math.rad(rot), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, 0) + targetHum.MoveDirection, CFrame.Angles(math.rad(rot), 0, 0))
                    task.wait()
                else
                    setFlingCFrame(targetPart, CFrame.new(0, 1.5, targetHum.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, -targetHum.WalkSpeed), CFrame.Angles(0, 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, 1.5, targetHum.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    if targetRoot then
                        setFlingCFrame(targetPart, CFrame.new(0, 1.5, targetRoot.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()
                        setFlingCFrame(targetPart, CFrame.new(0, -1.5, -targetRoot.Velocity.Magnitude / 1.25), CFrame.Angles(0, 0, 0))
                        task.wait()
                        setFlingCFrame(targetPart, CFrame.new(0, 1.5, targetRoot.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()
                    end
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(-90), 0, 0))
                    task.wait()
                    setFlingCFrame(targetPart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                    task.wait()
                end

                if targetPart.Velocity.Magnitude > 500 or targetPart.Parent ~= targetPlayer.Character or targetPlayer.Parent ~= Players or targetHum.Sit or hum.Health <= 0 or startTime + 2 < tick() then
                    break
                end
            end
        end

        Workspace.FallenPartsDestroyHeight = 0/0
        local bv = Instance.new("BodyVelocity")
        bv.Name = "EpixVel"
        bv.Parent = root
        bv.Velocity = Vector3.new(900000000, 900000000, 900000000)
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        if targetRoot then
            doFling(targetRoot)
        end

        if bv and bv.Parent then
            bv:Destroy()
        end

        hum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        if genv.OldPos and hum.Health > 0 then
            root.CFrame = genv.OldPos
        end
    end

    for _, v in ipairs(targetNames) do
        local target = findTarget(v)
        if allMode then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer then flingTarget(p) end
            end
            allMode = false
        elseif target then
            flingTarget(target)
        end
    end
end

-- Server Hop
ServerHop = function()
    local sList = {}
    pcall(function()
        local res = game:HttpGet(("https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=Asc&limit=100"):format(game.PlaceId))
        local data = HttpService:JSONDecode(res)
        local entries = data and data.data or {}
        for _, s in ipairs(entries) do
            if s.id ~= game.JobId and s.playing < s.maxPlayers then
                table.insert(sList, s.id)
            end
        end
    end)

    if #sList > 0 then
        TeleportService:TeleportToPlaceInstance(game.PlaceId, sList[math.random(1, #sList)])
    else
        pcall(function()
            if bc and bc.Notify then bc:Notify('No available servers found.') end
        end)
    end
end

-- Coin Farm 1: Fly Mode
local function GetClosestCoin()
    local root = GetRootPart()
    if not root then return nil end
    local list = {}
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model:FindFirstChild("CoinContainer") then
            for _, coin in ipairs(model.CoinContainer:GetChildren()) do
                if coin.Name == "Coin_Server" and not coin:FindFirstChild("CollectedCoin") and coin:FindFirstChild("TouchInterest") then
                    local dist = (root.Position - coin.Position).Magnitude
                    if dist < 100 then
                        table.insert(list, { Coin = coin, Distance = dist })
                    end
                end
            end
        end
    end
    table.sort(list, function(a, b) return a.Distance < b.Distance end)
    return list[1] and list[1].Coin
end

local coinFarm1Connections = {}
StopCoinFarm1 = function()
    for _, conn in ipairs(coinFarm1Connections) do
        if conn then pcall(function() conn:Disconnect() end) end
    end
    table.clear(coinFarm1Connections)

    local root = GetRootPart()
    if root then
        root.Anchored = false
        local vel = root:FindFirstChild("MM2_CoinFarmVelocity")
        if vel then vel:Destroy() end
    end
end

StartCoinFarm1 = function()
    StopCoinFarm1()

    local function stepFlyFarm()
        if not FarmSettings.CoinFarm.FlyEnabled then return end

        local mainGui = LocalPlayer.PlayerGui:FindFirstChild("MainGUI")
        local gameGui = mainGui and mainGui:FindFirstChild("Game")
        local ok, isBagVisible, isBagFull = pcall(function()
            return gameGui.CoinBags.Visible, gameGui.CoinBags.Container.Candy.Full.Visible
        end)
        local root = GetRootPart()

        if ok and isBagVisible and not isBagFull and root then
            if not root:FindFirstChild("MM2_CoinFarmVelocity") then
                local bv = Instance.new("BodyVelocity")
                bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                bv.Name = "MM2_CoinFarmVelocity"
                bv.Parent = root
            end

            DisableCollisions()

            local targetCoin = GetClosestCoin()
            if targetCoin then
                root.Anchored = false
                task.spawn(function()
                    local r = GetRootPart()
                    if not r then return end
                    local dist = (r.Position - targetCoin.Position).Magnitude
                    local duration = dist / math.max(FarmSettings.CoinFarm.FlySpeed, 1)
                    local targetCFrame = CFrame.new(targetCoin.Position + Vector3.new(0, -1, 0))

                    TweenService:Create(r, TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out), {
                        CFrame = targetCFrame
                    }):Play()

                    if targetCoin and targetCoin.Parent then
                        local touchConn
                        touchConn = targetCoin.Touched:Connect(function()
                            task.wait(0.1)
                            if targetCoin then
                                targetCoin:Destroy()
                                if touchConn then touchConn:Disconnect() end
                            end
                        end)
                    end
                end)
            else
                root.Anchored = true
            end

            task.wait(0.02)
        end
    end

    local hbConn = RunService.Heartbeat:Connect(stepFlyFarm)
    table.insert(coinFarm1Connections, hbConn)

    if CoinCollectedRemote then
        local collectConn = CoinCollectedRemote.OnClientEvent:Connect(function(_, current, max)
            if current == max then
                StopCoinFarm1()
            end
        end)
        table.insert(coinFarm1Connections, collectConn)
    end

    local hum = GetHumanoid()
    if hum then
        local diedConn = hum.Died:Connect(function()
            task.wait(1)
            repeat task.wait() until LocalPlayer.Character ~= nil
            repeat task.wait() until LocalPlayer.PlayerGui ~= nil
            task.wait(1)
            repeat task.wait() until LocalPlayer.PlayerGui:FindFirstChild("MainGUI")
            task.wait(1)
            TeleportToMap()
        end)
        table.insert(coinFarm1Connections, diedConn)
    end

    if RoundEndFadeRemote then
        local endConn = RoundEndFadeRemote.OnClientEvent:Connect(function()
            StopCoinFarm1()
        end)
        table.insert(coinFarm1Connections, endConn)
    end
end

local coinFarm2Running = false
StartCoinFarm2 = function()
    if coinFarm2Running then return end
    coinFarm2Running = true
    local isFull = false

    if FarmSettings.CoinFarm.SafeTPEnabled then
        TeleportToSafePlatform()
    end

    while not isFull do
        local map = GetMapContainer()
        if not map then
            task.wait(1)
            if FarmSettings.CoinFarm.SafeTPEnabled then
                continue
            end
        elseif FarmSettings.CoinFarm.SafeTPEnabled then
            task.spawn(function()
                pcall(DestroyMap)
            end)

            local lowestCoin = nil
            local lowestY = math.huge
            if map:FindFirstChild("CoinContainer") then
                for _, child in ipairs(map.CoinContainer:GetChildren()) do
                    if child:FindFirstChild("TouchInterest") and child.Position.Y < lowestY then
                        lowestY = child.Position.Y
                        lowestCoin = child
                    end
                end
            end

            if not lowestCoin then
                task.wait(0.1)
                continue
            end

            local coinCollected = false
            TeleportUnderCoin(lowestCoin)

            local collectConn
            if CoinCollectedRemote then
                collectConn = CoinCollectedRemote.OnClientEvent:Connect(function(_, current, max)
                    coinCollected = true
                    if current == max then
                        isFull = true
                    end
                end)
            end

            if isFull then
                if collectConn then collectConn:Disconnect() end
                break
            end

            while true do
                task.wait()
                task.spawn(function()
                    local root = GetRootPart()
                    if root and lowestCoin then
                        pcall(firetouchinterest, root, lowestCoin, 0)
                        pcall(firetouchinterest, root, lowestCoin, 1)
                    end
                end)
                task.wait()
                DisableCollisions()
                TeleportUnderCoin(lowestCoin)

                if coinCollected or not lowestCoin:FindFirstChild("TouchInterest") then
                    break
                end
            end

            if collectConn then collectConn:Disconnect() end
            TeleportToSafePlatform()
            task.wait(2)
            continue
        end
        break
    end

    coinFarm2Running = false
end
