-- Services
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TweenService = game:GetService("TweenService")

local LocalPlayer = Players.LocalPlayer

-- State Variables
local Config = {
    CoinAutofarm = false,
    AutoReset = false,
    AutoFling = false,
    FarmSpeed = 25
}

local RoundActive = false
local CurrentBagCount = 0
local MaxBagCapacity = 40 -- Will automatically adjust to 50 if Elite Gamepass is owned
local IsResetting = false
local StartPosition = nil
local TotalCollected = 0

-- Gamepass Check (Elite check for Bag Size)
local function CheckGamepass()
    local success, owns = pcall(function()
        return game:GetService("MarketplaceService"):UserOwnsGamePassAsync(LocalPlayer.UserId, 818078531)
    end)
    if success and owns then
        MaxBagCapacity = 50
    else
        MaxBagCapacity = 40
    end
end
CheckGamepass()

-- Helper: Get Humanoid Part Safely
local function GetRootPart()
    local char = LocalPlayer.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

-- Helper: Find Murderer
local function FindMurderer()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            if player.Character:FindFirstChild("Knife") or (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife")) then
                return player
            end
        end
    end
    return nil
end

-- Fling Method (Extracted from source core logic)
local function FlingPlayer(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    
    local myChar = LocalPlayer.Character
    local myHumanoid = myChar and myChar:FindFirstChildOfClass("Humanoid")
    local myRoot = GetRootPart()
    local targetRoot = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    
    if not (myChar and myHumanoid and myRoot and targetRoot) then return end
    
    local oldPos = myRoot.CFrame
    workspace.FallenPartsDestroyHeight = 0/0
    
    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.new(900000000, 900000000, 900000000)
    bv.MaxForce = Vector3.new(1/0, 1/0, 1/0)
    bv.Parent = myRoot
    
    myHumanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
    
    -- Fast loop targeted execution sequence
    local startFlingTime = tick()
    while myRoot and targetRoot and (tick() - startFlingTime < 2) do
        myRoot.CFrame = CFrame.new(targetRoot.Position) * CFrame.Angles(math.rad(90), 0, 0)
        myRoot.Velocity = Vector3.new(90000000, 900000000, 90000000)
        task.wait()
    end
    
    bv:Destroy()
    myHumanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    
    -- Return back safely
    myRoot.CFrame = oldPos
    workspace.FallenPartsDestroyHeight = -500
end

-- Scan for closest Items
local function GetClosestItem()
    local root = GetRootPart()
    if not root then return nil, math.huge end
    
    local closest = nil
    local minDistance = math.huge
    
    for _, container in pairs(workspace:GetChildren()) do
        if container:FindFirstChild("CoinContainer") then
            for _, item in pairs(container.CoinContainer:GetChildren()) do
                if item:IsA("BasePart") and item:FindFirstChild("TouchInterest") then
                    if item:GetAttribute("CoinID") ~= "Candy" then
                        local dist = (root.Position - item.Position).Magnitude
                        if dist < minDistance then minDistance = dist; closest = item end
                    end
                end
            end
        end
    end
    return closest, minDistance
end

-- Primary Farming Loop Thread
task.spawn(function()
    while true do
        if Config.CoinAutofarm and RoundActive and not IsResetting then
            local root = GetRootPart()
            local item, dist = GetClosestItem()
            
            if item and root then
                if dist > 150 then
                    root.CFrame = item.CFrame
                else
                    local tween = TweenService:Create(root, TweenInfo.new(dist / Config.FarmSpeed, Enum.EasingStyle.Linear), {CFrame = item.CFrame})
                    tween:Play()
                    repeat task.wait() until not (item:FindFirstChild("TouchInterest") and RoundActive and Config.CoinAutofarm)
                    tween:Cancel()
                end
                TotalCollected = TotalCollected + 1
            end
        end
        task.wait(0.2)
    end
end)

-- Anti-collision thread when farm is active
RunService.Stepped:Connect(function()
    if Config.CoinAutofarm and RoundActive and not IsResetting then
        local char = LocalPlayer.Character
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end
end)

-- Remote Hook listeners (Round states & collection updates)
local GameplayRemote = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("Gameplay")
if GameplayRemote then
    local CoinCollected = GameplayRemote:FindFirstChild("CoinCollected")
    local RoundStart = GameplayRemote:FindFirstChild("RoundStart")
    local RoundEnd = GameplayRemote:FindFirstChild("RoundEndFade")
    
    if RoundStart then
        RoundStart.OnClientEvent:Connect(function()
            RoundActive = true
            local root = GetRootPart()
            if root then StartPosition = root.CFrame end
        end)
    end
    
    if RoundEnd then
        RoundEnd.OnClientEvent:Connect(function()
            RoundActive = false
        end)
    end
    
    if CoinCollected then
        CoinCollected.OnClientEvent:Connect(function(itemType, scoreAmount)
            if itemType == "Coin" then
                CurrentBagCount = scoreAmount
                
                -- Actions when the Inventory Bag hits full capacity limits
                if CurrentBagCount >= MaxBagCapacity then
                    Config.CoinAutofarm = false
                    
                    -- Handle auto fling target
                    if Config.AutoFling then
                        local mud = FindMurderer()
                        if mud then FlingPlayer(mud) end
                    end
                    
                    -- Handle auto reset character routine
                    if Config.AutoReset and not IsResetting then
                        IsResetting = true
                        local root = GetRootPart()
                        if root and StartPosition then
                            root.CFrame = StartPosition
                            task.wait(0.5)
                        end
                        if LocalPlayer.Character then
                            LocalPlayer.Character:BreakJoints()
                        end
                    end
                end
            end
        end)
    end
end

LocalPlayer.CharacterAdded:Connect(function()
    CurrentBagCount = 0
    IsResetting = false
    CheckGamepass()
end)
