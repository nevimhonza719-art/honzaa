    -- Coin skipping & failed tracker
    local coinFails = {}
    local coinSkips = {}

    local function coinSkip(coin)
        local t = coinSkips[coin]
        if not t then return false end
        if t < tick() then
            coinSkips[coin] = nil
            coinFails[coin] = nil
            return false
        end
        return true
    end

    local function coinFailed(coin)
        local n = (coinFails[coin] or 0) + 1
        coinFails[coin] = n
        if n >= 2 then
            coinSkips[coin] = tick() + 20
        end
    end

    local function coinReset()
        coinFails = {}
        coinSkips = {}
    end

    local function getClosestCoin(currentMap)
        local root = getRoot(localPlayer)
        if not root or not currentMap then return nil end
        local container = currentMap:FindFirstChild("CoinContainer")
        if not container then return nil end
        local closest, closestDist = nil, math.huge
        for _, coin in ipairs(container:GetChildren()) do
            if coin.Name == "Coin_Server" and not coin:GetAttribute("Collected") and not coinSkip(coin) then
                local dist = (root.Position - coin.Position).Magnitude
                if dist < closestDist then
                    closestDist = dist
                    closest = coin
                end
            end
        end
        return closest
    end

    local function getBagCoins()

                container = localPlayer.PlayerGui.MainGUI.Lobby.Dock.CoinBags.Container

            for _, obj in ipairs(container:GetChildren()) do
                if obj:IsA("GuiObject") and obj:FindFirstChild("CurrencyFrame") then
                    if obj.Visible then
                        activeBag = obj
                        break
                    elseif obj.Name == "Coin" and not activeBag then
                        activeBag = obj

            if not activeBag then return 0 end
            local full = activeBag:FindFirstChild("Full")
            if full and full.Visible then return 999 end

            local icon = currFrame and currFrame:FindFirstChild("Icon")

            local txt = coinsLabel and coinsLabel.Text
            if txt and (txt:lower():find("full") or txt:lower():find("max")) then
                return 999
            end
            return tonumber(txt) or 0
        end)
        return (ok and res) or 0
    end

        currentMap = currentMap or getCurrentMap()
        if not currentMap then return end

        if not spawns then return end
        local chosenSpawn = nil
        for _, sp in ipairs(spawns:GetChildren()) do
            if sp:IsA("BasePart") then
                if sp.Name == "Spawn" or sp.Name == "PlayerSpawn" or sp.Name == "SpawnLocation" then
                    chosenSpawn = sp

        local hum = getHumanoid(localPlayer)
        if root then
            if hum then
                hum.PlatformStand = false
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)

            root.RotVelocity = Vector3.zero
            root.CFrame = CFrame.new(chosenSpawn.Position.X, chosenSpawn.Position.Y + chosenSpawn.Size.Y / 2 + 4, chosenSpawn.Position.Z)

    end

    -- SkidFling
    local flingActive = false
    local function flingPlayer(targetPlayer)
        if flingActive then
            notify("Fling Active!", "Wait for the current fling to finish!", 2)
            return
        end
        if not targetPlayer or not targetPlayer.Character then
            notify("Error!", "Could not find that player!", 2)
            return
        end
        local targetHum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
        if not targetHum or targetHum.Sit then return end
        local char = localPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = hum and hum.RootPart
        if not root then return end

        flingActive = true
        local originalCF = root.CFrame
        local targetRoot = targetHum.RootPart
        local targetHead = targetPlayer.Character:FindFirstChild("Head")

        local function flingMove(part, offset, angles)
            root.CFrame = CFrame.new(part.Position) * offset * angles
            pcall(function() char:SetPrimaryPartCFrame(CFrame.new(part.Position) * offset * angles) end)
            root.Velocity = Vector3.new(90000000, 900000000, 90000000)
            root.RotVelocity = Vector3.new(900000000, 900000000, 900000000)
        end

        local bv = Instance.new("BodyVelocity")
        bv.Name = "SkidFlingBV"
        bv.Parent = root
        bv.Velocity = Vector3.zero
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        local targetPart = targetRoot or targetHead
        if targetPart then
            local startT = tick()
            local rotAngle = 0
            while flingActive and (tick() - startT < 2.5) do
                if not targetPart or not targetPart.Parent or not targetPlayer.Character then break end
                if targetHum.Health <= 0 or hum.Health <= 0 then break end
                if targetPart.Velocity.Magnitude > 500 then break end

                rotAngle = rotAngle + 100
                local md = targetHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25
                flingMove(targetPart, CFrame.new(0, 1.5, 0) + md, CFrame.Angles(math.rad(rotAngle), 0, 0))
                task.wait()
                flingMove(targetPart, CFrame.new(0, -1.5, 0) + md, CFrame.Angles(math.rad(rotAngle), 0, 0))
                task.wait()
                flingMove(targetPart, CFrame.new(2.25, 1.5, -2.25) + md, CFrame.Angles(math.rad(rotAngle), 0, 0))
                task.wait()
                flingMove(targetPart, CFrame.new(-2.25, -1.5, 2.25) + md, CFrame.Angles(math.rad(rotAngle), 0, 0))
                task.wait()
            end
        end

        bv:Destroy()
        hum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)

        if originalCF then
            for i = 1, 30 do
                root.CFrame = originalCF * CFrame.new(0, 0.5, 0)
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                for _, c in ipairs(char:GetChildren()) do
                    if c:IsA("BasePart") then
                        c.Velocity = Vector3.zero
                        c.RotVelocity = Vector3.zero
                    end
                end
                task.wait()
                if (root.Position - originalCF.Position).Magnitude < 25 then break end
            end
        end

        flingActive = false
    end

                local hum = p.Character:FindFirstChildOfClass("Humanoid")

                end
            end
        end
        return count
    end

    -- Server Hop Watcher & Populated Hop
    local hopping = false
    local function populatedHop()
        if hopping then return end
        hopping = true
        notify("Server Empty", "Hopping to populated server...", 3)
        task.wait(0.5)

        local bestServer = nil
        local cursor = ""
        local attempts = 0

        local function httpGetJson(url)
            local ok, body = pcall(function()
                if syn and syn.request then return syn.request({ Url = url, Method = "GET" }).Body end
                if request then return request({ Url = url, Method = "GET" }).Body end
                if http_request then return http_request({ Url = url, Method = "GET" }).Body end
                return game:HttpGet(url)
            end)
            if ok and body then
                local decOk, data = pcall(HttpService.JSONDecode, HttpService, body)
                return decOk and data or nil
            end
            return nil
        end

        while attempts < 4 do
            attempts = attempts + 1
            local url = "https://games.roblox.com/v1/games/" .. placeId .. "/servers/Public?sortOrder=Desc&limit=100"
            if cursor ~= "" then url = url .. "&cursor=" .. cursor end
            local res = httpGetJson(url)
            if res and res.data then
                for _, s in ipairs(res.data) do
                    if s.id ~= jobId and s.playing and s.maxPlayers and s.playing < s.maxPlayers then
                        if not bestServer or s.playing > bestServer.playing then
                            bestServer = s
                        end
                    end
                end
                cursor = res.nextPageCursor or ""
                if cursor == "" then break end
            else
                break
            end
        end

        if bestServer then
            notify("Hopping", "Joining server with " .. bestServer.playing .. " players...", 3)
            task.wait(0.5)
            pcall(function()
                TeleportService:TeleportToPlaceInstance(placeId, bestServer.id)
            end)
        else
            pcall(function()
                TeleportService:Teleport(placeId)
            end)
        end
        task.wait(10)
        hopping = false
    end

    local hopWatchThread = nil

        if hopWatchThread then return end
        hopWatchThread = task.spawn(function()
            local lowPlayerSeconds = 0
            while Toggles and Toggles.CoinAutofarm and Toggles.CoinAutofarm.Value do
                task.wait(3)

                    lowPlayerSeconds = lowPlayerSeconds + 3

                        populatedHop()
                        lowPlayerSeconds = 0
                    end
                else
                    lowPlayerSeconds = 0
                end
            end
            hopWatchThread = nil
        end)
    end

    -- Discord Webhook Sender
    local function sendWebhookPayload(url, jsonBody)
        local req = syn and syn.request or request or http_request or (http and http.request)
        if not req then return false, "No request function" end
        local ok, res = pcall(req, {
            Url = url,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = jsonBody,
        })
        if not ok then return false, "Request error" end
        local code = tonumber(type(res) == "table" and (res.StatusCode or res.Status)) or 0
        return (code == 200 or code == 204), tostring(code)
    end

    local function postDiscordEmbed(title, description, fields, thumb)
        local url = Options and Options.DiscordWebhook and Options.DiscordWebhook.Value or ""
        url = url:gsub("^%s+", ""):gsub("%s+$", "")

            username = "Creep.cc Autofarm",
            embeds = { embed }
        })

        task.spawn(function()
            sendWebhookPayload(url, payload)
        end)
    end

    -- Performance Mode
    local perfStash = { parts = {}, lighting = nil }
    setPerfMode = function(enable)
        if enable then
            if not perfStash.lighting then
                perfStash.lighting = {
                    GlobalShadows = Lighting.GlobalShadows,
                    ShadowSoftness = Lighting.ShadowSoftness,
                }
                Lighting.GlobalShadows = false
                Lighting.ShadowSoftness = 0
            end
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("BasePart") and not (localPlayer.Character and obj:IsDescendantOf(localPlayer.Character)) then
                    if not perfStash.parts[obj] then
                        perfStash.parts[obj] = { Material = obj.Material, CastShadow = obj.CastShadow }
                        obj.Material = Enum.Material.SmoothPlastic
                        obj.CastShadow = false
                    end
                elseif obj:IsA("Decal") or obj:IsA("Texture") then
                    obj.Transparency = 1
                elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                    obj.Enabled = false
                end
            end
        else
            if perfStash.lighting then
                Lighting.GlobalShadows = perfStash.lighting.GlobalShadows
                Lighting.ShadowSoftness = perfStash.lighting.ShadowSoftness
                perfStash.lighting = nil
            end
            for part, data in pairs(perfStash.parts) do
                if part and part.Parent then
                    pcall(function()
                        part.Material = data.Material
                        part.CastShadow = data.CastShadow
                    end)
                end
            end
            perfStash.parts = {}
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("Decal") or obj:IsA("Texture") then
                    obj.Transparency = 0
                elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                    obj.Enabled = true
                end
            end
        end
    end

        if not ok or type(prof) ~= "table" then return nil end
        local mat = prof.Materials and prof.Materials.Owned
        return type(mat) == "table" and mat or nil
    end

    local function getBoxPrice(boxName)
        local cur = BOX_CURRENCY[boxName] or "Coins"
        local ok, shop = pcall(require, ReplicatedStorage.Database.Sync.NewShop)
        if not ok or type(shop) ~= "table" then return cur, nil end
        local item = shop[boxName]
        local price = type(item) == "table" and type(item.Price) == "table" and item.Price or nil
        return cur, price and tonumber(price[cur]) or nil
    end

    local function getBoxFunds(curName)
        local owned = getOwnedMaterials()
        if not owned then return nil end
        return tonumber(owned[curName]) or 0
    end

    local function openBoxRemote(boxName)
        local rem = ReplicatedStorage:FindFirstChild("Remotes")
        rem = rem and rem:FindFirstChild("Shop")
        rem = rem and rem:FindFirstChild("OpenCrate")
        if not rem then return nil, "OpenCrate remote missing" end
        local cur = BOX_CURRENCY[boxName] or "Coins"
        local ok, res = pcall(function()
            return rem:InvokeServer(boxName, "MysteryBox", cur)
        end)
        if not ok then return nil, tostring(res) end
        return res
    end

    local activeBoxLoops = {}
    local function startBoxLoop(boxId, boxName, toggleName)
        if activeBoxLoops[boxId] then return end
        activeBoxLoops[boxId] = true

        task.spawn(function()
            local openedCount = 0
            while Toggles and Toggles[toggleName] and Toggles[toggleName].Value do
                local cur, price = getBoxPrice(boxId)
                local funds = getBoxFunds(cur)
                if price and funds and funds < price then
                    task.wait(5)
                else
                    local res, err = openBoxRemote(boxId)
                    if not res then
                        local f = getBoxFunds(cur)
                        if price and f and f < price then
                            task.wait(5)
                        else
                            if Toggles[toggleName] then Toggles[toggleName]:SetValue(false) end
                            notify(boxName, "Stopped: " .. (err or "server refused"), 4)
                            break
                        end
                    else
                        openedCount = openedCount + 1
                        if Toggles and Toggles.DiscordSender and Toggles.DiscordSender.Value then
                            postDiscordEmbed("Opened " .. boxName, "Successfully opened " .. boxName .. " (#" .. openedCount .. ")", {
                                { name = "Box", value = boxName, inline = true },
                                { name = "Reward", value = tostring(res), inline = true },
                            })
                        end
                        task.wait(1)
                    end
                end
            end
            activeBoxLoops[boxId] = nil
        end)
    end

    -- Main Autofarm State & Loop
    local farmRunning = false
    local farmTween = nil
    local lastMap = nil
    local sawCoins = false
    local flingDoneFired = false
    local killFired = false
    local targetCoin = nil
    local farmTargetTime = tick()
    local stuckCoin = nil
    local stuckTime = 0

    stopFarm = function()
        farmRunning = false
        if farmTween then
            pcall(function() farmTween:Cancel() end)
            farmTween = nil
        end
        setFarmFloat(false)
        if inActiveRound() then
            pcall(teleportHome)
        end
        task.spawn(function()
            task.wait(0.35)
            setFarmFloat(false)
            if inActiveRound() then pcall(teleportHome) end
        end)
    end

    local function finishFarmAction()
        if farmTween then
            farmTween:Cancel()
            farmTween = nil
        end
        setFarmFloat(false)

        -- Fling Murderer
        if Toggles and Toggles.FlingMurdererWhenDone and Toggles.FlingMurdererWhenDone.Value and not flingDoneFired then
            local murd = getMurderer()
            if murd and murd ~= localPlayer then
                flingDoneFired = true
                task.spawn(function()
                    if Toggles.AutoResetWhenBagFull and Toggles.AutoResetWhenBagFull.Value then
                        local hum = getHumanoid(localPlayer)
                        if hum then hum.Health = 0 end
                        local startT = tick()
                        while tick() - startT < 10 do
                            task.wait(0.2)
                            local c = localPlayer.Character
                            local h = c and c:FindFirstChildOfClass("Humanoid")
                            if h and h.Health > 0 and c:FindFirstChild("HumanoidRootPart") then break end
                        end
                        task.wait(0.6)
                        murd = getMurderer()
                    end
                    if murd and murd ~= localPlayer then
                        pcall(flingPlayer, murd)
                    end
                end)
            end
        end

        -- Kill All When Full
        if Toggles and Toggles.KillAllWhenBagFull and Toggles.KillAllWhenBagFull.Value and not killFired then
            if getMurderer() == localPlayer then
                killFired = true
                task.spawn(function()
                    pcall(killAll)
                end)
            end
        end
    end

    startFarm = function()
        if farmRunning then return end
        farmRunning = true
        startFarmHopWatch()

        task.spawn(function()
            while farmRunning and Toggles and Toggles.CoinAutofarm and Toggles.CoinAutofarm.Value do
                task.wait(0.1)
                pcall(function()
                    if not farmRunning then return end
                    local root = getRoot(localPlayer)
                    local hum = getHumanoid(localPlayer)
                    if not root or not hum or hum.Health <= 0 then
                        return
                    end

                    local curMap = getCurrentMap()
                    if curMap ~= lastMap then
                        lastMap = curMap
                        sawCoins = false
                        flingDoneFired = false
                        killFired = false
                        coinReset()
                    end

                    if not curMap then
                        setFarmFloat(false)
                        return
                    end

                    -- Check round timer (if round not active, stand on spawn)
                    if getRoundTimer() <= 0 then
                        setFarmFloat(false)
                        return
                    end

                    -- Check bag capacity (40 normal, 50 elite)
                    local maxBag = localPlayer:GetAttribute("Elite") and 50 or 40
                    local bagCoins = getBagCoins()
                    if bagCoins >= maxBag or bagCoins >= 999 then
                        finishFarmAction()
                        return
                    end

                    -- Find next coin (before coins spawn, stand where spawned)
                    local coin = getClosestCoin(curMap)
                    if coin then
                        sawCoins = true
                        if coin ~= targetCoin or (targetCoin and targetCoin:GetAttribute("Collected")) then
                            targetCoin = coin
                            farmTargetTime = tick()
                            if farmTween then
                                farmTween:Cancel()
                                farmTween = nil
                            end
                            setFarmFloat(true)

                            local farmSpeed = (Options and Options.FarmSpeed and Options.FarmSpeed.Value) or 20
                            local dist = (root.Position - coin.Position).Magnitude
                            local duration = math.clamp(dist / farmSpeed, 0.05, 15)

                            root.Velocity = Vector3.zero
                            local coinCF = coin.CFrame
                            local targetCF = CFrame.new(coinCF.X, coinCF.Y - 3.5, coinCF.Z) * CFrame.Angles(math.pi/2, 0, math.pi/2)

                            farmTween = TweenService:Create(root, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
                                CFrame = targetCF
                            })
                            farmTween:Play()
                            root.Velocity = Vector3.zero
                        else
                            local dist = (root.Position - coin.Position).Magnitude
                            if dist <= 6 and not coin:GetAttribute("Collected") then
                                coin.CFrame = CFrame.new(root.Position + Vector3.new(math.random(-80, 80)/100, math.random(-50, 300)/100, math.random(-80, 80)/100))
                                if (coin.Position - root.Position).Magnitude > 0.1 then
                                    root.CFrame = root.CFrame * CFrame.new(0, math.random(-20, 20)/100, math.random(-20, 20)/100)
                                end
                                if stuckCoin ~= coin then
                                    stuckCoin = coin
                                    stuckTime = tick()
                                end
                                if tick() - stuckTime > 2 then
                                    stuckCoin = nil
                                    stuckTime = 0
                                    targetCoin = nil
                                end
                            elseif dist > 6 then
                                if tick() - farmTargetTime > 4 then
                                    coinFailed(coin)
                                    targetCoin = nil
                                    stuckCoin = nil
                                    stuckTime = 0
                                end
                            end
                        end
                    else
                        if sawCoins then
                            finishFarmAction()
                        end
                    end
                end)
            end
            stopFarm()
        end)
    end

    -- Mid-round death handler & round event signals
    local diedMidRound = false
    task.spawn(function()
        localPlayer.CharacterAdded:Connect(function(char)
            local hum = char:WaitForChild("Humanoid", 5)
            if hum then
                hum.Died:Connect(function()
                    local murd = getMurderer()
                    diedMidRound = (murd ~= nil and murd ~= localPlayer)
                end)
            end

            if diedMidRound and Toggles and Toggles.FlingMurdererWhenDone and Toggles.FlingMurdererWhenDone.Value and not flingDoneFired then
                diedMidRound = false
                task.spawn(function()
                    task.wait(1.2)
                    local murd = getMurderer()
                    if murd and murd ~= localPlayer then
                        flingDoneFired = true
                        notify("Out Of The Round", "Flinging " .. murd.Name .. "...", 3)
                        pcall(flingPlayer, murd)
                    end
                end)
            end
        end)

        local remotes = ReplicatedStorage:WaitForChild("Remotes", 10)
        local gameplay = remotes and remotes:WaitForChild("Gameplay", 10)
        if gameplay then
            local victoryScreen = gameplay:FindFirstChild("VictoryScreen")
            if victoryScreen then
                victoryScreen.OnClientEvent:Connect(function(p1, roundRole, roundWin)
                    if Toggles and Toggles.DiscordSender and Toggles.DiscordSender.Value then
                        local fields = {
                            { name = "Round Outcome", value = tostring(roundWin or "Over"), inline = true },
                            { name = "Final Role", value = tostring(roundRole or "Unknown"), inline = true },
                        }
                        postDiscordEmbed("Round Finished", "The game round has ended.", fields)
                    end
                end)
            end
        end
    end)

    -- UI SECTION: Tab & Groupboxes
    local FarmMain = ch.Autofarm:AddLeftGroupbox('Coin Autofarm')
    local FarmSettings = ch.Autofarm:AddRightGroupbox('Settings')
    local FarmWebhook = ch.Autofarm:AddLeftGroupbox('Discord Webhook')

    FarmMain:AddSlider('FarmSpeed', {

    })

    -- Right: Settings
    FarmSettings:AddToggle('FarmPerfMaster', {
        Text = 'Performance Mode',

    FarmSettings:AddToggle('FarmDeadHop', {

    })

    FarmSettings:AddToggle('AntiAFK', {
        Text = 'Anti-AFK',
        Default = true,
        Tooltip = 'Prevents you from getting AFK kicked.',
        Callback = function(Value)

        end,

        Text = 'Auto-Reset When Bag Full',

            setAutoReset(Value)

    })

    FarmSettings:AddToggle('AutoClaimShells', {
        Text = 'Auto Claim Shells',

        Callback = function(Value)

        Text = 'Kill All When Bag Full',

    })

    -- Left: Discord Webhook
    FarmWebhook:AddInput('DiscordWebhook', {

    FarmWebhook:AddToggle('DiscordSender', {
        Text = 'Start Sender',

                    { name = "User ID", value = tostring(localPlayer.UserId), inline = true },

        end,
    })

    FarmWebhook:AddButton('Test Webhook', function()
        postDiscordEmbed("Test Webhook", "This is a test notification from Creep.cc Autofarm.")
        notify("Test Sent", "Sent test embed to webhook.", 2)
    end)

    -- Right: Auto Open Boxes
    for _, box in ipairs(BOXES) do
        local boxId = box[1]
        local boxTitle = box[2]

            Tooltip = 'Automatically opens ' .. boxTitle .. ' when affordable.',

                end

    -- Auto-start farm on load if enabled
    task.spawn(function()
        task.wait(0.25)
        if Toggles and Toggles.CoinAutofarm and Toggles.CoinAutofarm.Value then
            startFarm()
        end
    end)
